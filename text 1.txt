local function drawIcon(parent,kind,size)

local function fill(dx,dy,w,hh,r,rot)
local f=new("Frame",{AnchorPoint=Vector2.new(0.5,0.5),Position=UDim2.new(0.5,dx,0.5,dy),Size=UDim2.fromOffset(w,hh),BorderSizePixel=0,Rotation=rot or 0,Parent=h})

local function ring(w)
local f=new("Frame",{AnchorPoint=Vector2.new(0.5,0.5),Position=UDim2.fromScale(0.5,0.5),Size=UDim2.fromOffset(w,w),BackgroundTransparency=1,Parent=h})

strokes[#strokes+1]=new("UIStroke",{Thickness=math.max(1,thin-1),Parent=f})
end
if kind=="eye"then ring(s*0.85);fill(0,0,s*0.34,s*0.34,full)
elseif kind=="ring"then ring(s*0.8)
elseif kind=="diamond"then fill(0,0,s*0.62,s*0.62,UDim.new(0,2),45)
elseif kind=="dots"then for _,d in ipairs({{-1,-1},{1,-1},{-1,1},{1,1}})do fill(d[1]*s*0.22,d[2]*s*0.22,s*0.32,s*0.32,full)end
elseif kind=="bars"then for i=-1,1 do fill(0,i*s*0.3,s*0.85,thin,full)end
elseif kind=="plus"then fill(0,0,s*0.85,thin,full);fill(0,0,thin,s*0.85,full)
elseif kind=="x"then fill(0,0,s*0.85,thin,full,45);fill(0,0,s*0.85,thin,full,-45)
elseif kind=="kebab"then for i=-1,1 do fill(0,i*s*0.32,thin+1,thin+1,full)end

local lens=new("Frame",{AnchorPoint=Vector2.new(0.5,0.5),Position=UDim2.new(0.5,-s*0.1,0.5,-s*0.1),Size=UDim2.fromOffset(s*0.58,s*0.58),BackgroundTransparency=1,Parent=h})

elseif kind=="chevron"then fill(-s*0.16,0,s*0.5,thin,full,45);fill(s*0.16,0,s*0.5,thin,full,-45)
else fill(0,0,s*0.5,s*0.5,full)end
return function(color)for _,p in ipairs(fills)do p.BackgroundColor3=color end;for _,st in ipairs(strokes)do st.Color=color end end
end

local function applyShard(piece,tau)
local t=math.max(0,tau-piece.delay)
local crack=math.clamp(t/0.12,0,1)
local fly=math.max(0,t-0.12)
local flyProgress=math.clamp(fly/0.6,0,1)
local offset=piece.direction*5*crack+piece.velocity*fly+Vector2.new(0,0.5*900*fly*fly)
piece.shard.Position=UDim2.fromOffset(piece.home.X+offset.X,piece.home.Y+offset.Y)
piece.shard.Rotation=piece.crackRotation*crack+piece.spin*fly
piece.depth.Scale=1-(1-piece.farScale)*flyProgress
piece.edge.Transparency=1-0.75*crack
piece.shard.GroupTransparency=math.clamp((flyProgress-0.35)/0.65,0,1)
end
local function createShards()
clearShards()
shardLayer.Position=frame.Position
shardScale.Scale=windowScale.Scale
shardLayer.Visible=true
local snapshot=makeSnapshot()
for col=1,2 do
local x0=(col-1)*300;local x1=col*300
local home=Vector2.new(x0+150,200)
local shard=Instance.new("CanvasGroup")
shard.AnchorPoint=Vector2.new(0.5,0.5)
shard.Position=UDim2.fromOffset(home.X,home.Y)
shard.Size=UDim2.fromOffset(300,400)
shard.BackgroundTransparency=1;shard.Parent=shardLayer
local copy=snapshot:Clone();copy.Position=UDim2.fromOffset(-x0,0);copy.Parent=shard
local edge=Instance.new("UIStroke");edge.Color=C.accent;edge.Thickness=1.5;edge.Transparency=1;edge.Parent=shard
local depth=Instance.new("UIScale");depth.Parent=shard
local direction=home-Vector2.new(300,200);direction=direction.Magnitude>1 and direction.Unit or Vector2.new(0,-1)
table.insert(shardPieces,{shard=shard,edge=edge,depth=depth,home=home,direction=direction,crackRotation=(math.random()-0.5)*5,velocity=direction*(260+math.random()*80)+Vector2.new(0,-40-math.random()*60),spin=(direction.X<0 and-1 or 1)*(25+math.random()*20),farScale=0.3+math.random()*0.25,delay=math.random()*0.07})
end
end
local function openMenu()
animationId+=1;local id=animationId;menuOpen=true
local a=screenGui.AbsoluteSize
if a.X>1 and a.Y>1 then windowScale.Scale=math.clamp(math.min((a.X-48)/600,(a.Y-48)/400),0.45,1)end
frame.Visible=false;screenGui.Enabled=true;createShards()
for _,piece in ipairs(shardPieces)do applyShard(piece,TOTAL_TIME)end
local started=os.clock()
shardConnection=RunService.RenderStepped:Connect(function()
local u=math.clamp((os.clock()-started)/0.6,0,1);local tau=TOTAL_TIME*(1-u)^2
for _,piece in ipairs(shardPieces)do applyShard(piece,tau)end
if u>=1 then clearShards();if animationId==id then frame.Visible=true end end
end)
end
local function closeMenu()
animationId+=1;local id=animationId;menuOpen=false;createShards();frame.Visible=false
local started=os.clock()
shardConnection=RunService.RenderStepped:Connect(function()
local tau=os.clock()-started
for _,piece in ipairs(shardPieces)do applyShard(piece,tau)end
if tau>=TOTAL_TIME then clearShards();if animationId==id then screenGui.Enabled=false;frame.Visible=true end end
end)
end
UserInputService.InputBegan:Connect(function(input,gp)
if gp then return end

end
end)
